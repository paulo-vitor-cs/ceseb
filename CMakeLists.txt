cmake_minimum_required(VERSION 3.28)
project(ceseb VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(CESEB_BUILD_GUI "Build the Qt Widgets application" ON)
option(CESEB_BUILD_TESTS "Build the independent contract tests" ON)
option(CESEB_ENABLE_COVERAGE "Enable optional GNU coverage diagnostics" OFF)
set(CESEB_SOURCE_REVISION "" CACHE STRING "Real source commit for an exported snapshot")
if(CESEB_SOURCE_REVISION STREQUAL "")
  find_package(Git QUIET)
  if(Git_FOUND)
    execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse HEAD WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" OUTPUT_VARIABLE CESEB_SOURCE_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE _git_status)
  endif()
endif()
string(LENGTH "${CESEB_SOURCE_REVISION}" _revision_length)
if(NOT _revision_length EQUAL 40 OR NOT CESEB_SOURCE_REVISION MATCHES "^[0-9a-fA-F]+$")
  message(FATAL_ERROR "Provide the real 40-character CESEB_SOURCE_REVISION for this snapshot")
endif()
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/generated/ceseb")
configure_file(cmake/Version.hpp.in "${CMAKE_BINARY_DIR}/generated/ceseb/version.hpp" @ONLY)
include(cmake/Dependencies.cmake)
add_library(ceseb_core STATIC src/core/finance.cpp src/core/exact_math.cpp)
target_include_directories(ceseb_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(ceseb_core PUBLIC ceseb_boost_headers)
target_compile_features(ceseb_core PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(ceseb_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(ceseb_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(CESEB_ENABLE_COVERAGE)
  if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    message(FATAL_ERROR "Coverage diagnostics are selected only for GNU; no alternative instrument is inferred")
  endif()
  target_compile_options(ceseb_core PUBLIC --coverage -O0 -g)
  target_link_options(ceseb_core PUBLIC --coverage)
endif()
if(CESEB_BUILD_GUI)
  find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS Core Gui Widgets)
  add_library(ceseb_ui STATIC src/ui/money_codec.cpp src/ui/simulation_controller.cpp src/ui/main_window.cpp resources/application.qrc)
  set_target_properties(ceseb_ui PROPERTIES AUTOMOC ON AUTORCC ON)
  target_include_directories(ceseb_ui PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src" PRIVATE "${CMAKE_BINARY_DIR}/generated")
  target_link_libraries(ceseb_ui PUBLIC ceseb_core Qt6::Core Qt6::Gui Qt6::Widgets)
  if(MSVC)
    target_compile_options(ceseb_ui PRIVATE /W4 /utf-8)
  endif()
  add_executable(ceseb WIN32 src/app/main.cpp)
  target_include_directories(ceseb PRIVATE "${CMAKE_BINARY_DIR}/generated")
  target_link_libraries(ceseb PRIVATE ceseb_ui)
  set_target_properties(ceseb PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/cmake/Packaging.cmake")
    include(cmake/Packaging.cmake)
  endif()
endif()
if(CESEB_BUILD_TESTS)
  enable_testing()
  find_package(GTest 1.17.0 EXACT CONFIG REQUIRED)
  add_subdirectory(tests)
endif()
